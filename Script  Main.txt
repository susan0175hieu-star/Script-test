-- [[ UI LIBRARY HELPER ]] --
local function Create(className, props)
    local inst = Instance.new(className)
    for k, v in pairs(props) do
        inst[k] = v
    end
    return inst
end

local function CreateCorner(radius, parent)
    return Create("UICorner", { CornerRadius = UDim.new(0, radius), Parent = parent })
end

local function CreateStroke(thickness, color, parent)
    return Create("UIStroke", { Thickness = thickness, Color = color, ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Parent = parent })
end

local function AddGradient(parent, color1, color2)
    return Create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, color1),
            ColorSequenceKeypoint.new(1, color2)
        }),
        Rotation = 45,
        Parent = parent
    })
end

local function Bind(signal, callback)
    local con = signal:Connect(callback)
    table.insert(ConnectionTable, con)
    return con
end

local function CreateCombatToggle(text, callback, activeColor)
    activeColor = activeColor or COLORS.SELECTED
    local frame = Create("Frame", {
        Size = UDim2.new(1, 0, 0, 45),
        BackgroundColor3 = COLORS.BUTTON,
        Parent = CombatContainer
    })
    CreateCorner(6, frame)
    local stroke = CreateStroke(1, Color3.fromRGB(70, 70, 75), frame)

    local label = Create("TextLabel", {
        Text = text,
        Size = UDim2.new(1, -70, 1, 0),
        Position = UDim2.new(0, 15, 0, 0),
        BackgroundTransparency = 1,
        TextColor3 = COLORS.TEXT,
        Font = Enum.Font.GothamBold,
        TextSize = 14,
        TextXAlignment = Enum.TextXAlignment.Left,
        Parent = frame
    })

    local switchBG = Create("Frame", {
        Size = UDim2.new(0, 44, 0, 22),
        Position = UDim2.new(1, -55, 0.5, -11),
        BackgroundColor3 = Color3.fromRGB(30, 30, 35),
        Parent = frame
    })
    CreateCorner(11, switchBG)
    local switchStroke = CreateStroke(2, Color3.fromRGB(50, 50, 55), switchBG)

    local knob = Create("Frame", {
        Size = UDim2.new(0, 18, 0, 18),
        Position = UDim2.new(0, 2, 0.5, -9),
        BackgroundColor3 = Color3.fromRGB(180, 180, 180),
        Parent = switchBG
    })
    CreateCorner(10, knob)

    local btn = Create("TextButton", {
        Text = "",
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        Parent = frame
    })

    local function Update(animate)
        local state = (typeof(callback(nil)) == "boolean") and callback(nil) or false
        local targetPos = state and UDim2.new(1, -20, 0.5, -9) or UDim2.new(0, 2, 0.5, -9)
        local targetColor = state and COLORS.SELECTED or Color3.fromRGB(30, 30, 35)
        local knobColor = state and Color3.new(1, 1, 1) or Color3.fromRGB(180, 180, 180)

        if animate then
            TweenService:Create(knob, TweenInfo.new(0.2), {Position = targetPos, BackgroundColor3 = knobColor}):Play()
            TweenService:Create(switchBG, TweenInfo.new(0.2), {BackgroundColor3 = targetColor}):Play()
            TweenService:Create(switchStroke, TweenInfo.new(0.2), {Color = state and (activeColor == COLORS.SELECTED and Color3.new(1, 1, 1) or activeColor) or Color3.fromRGB(50, 50, 55), Transparency = state and 0.5 or 0}):Play()
            TweenService:Create(stroke, TweenInfo.new(0.2), {Color = state and activeColor or Color3.fromRGB(70, 70, 75)}):Play()
        else
            knob.Position = targetPos
            knob.BackgroundColor3 = knobColor
            switchBG.BackgroundColor3 = targetColor
            switchStroke.Color = state and (activeColor == COLORS.SELECTED and Color3.new(1, 1, 1) or activeColor) or Color3.fromRGB(50, 50, 55)
            switchStroke.Transparency = state and 0.5 or 0
            stroke.Color = state and activeColor or Color3.fromRGB(70, 70, 75)
        end
    end

    Bind(btn.MouseButton1Click, function()
        callback(true)
        Update(true)
    end)
    Update(false)
end

local function CreateCombatSlider(name, min, max, init, callback)
    local frame = Create("Frame", {
        Size = UDim2.new(1, 0, 0, 45),
        BackgroundTransparency = 1,
        Parent = CombatContainer
    })

    local dot = Create("Frame", {

    })
    CreateCorner(2, dot)

    local inner = Create("Frame", {
        Size = UDim2.new(1, -15, 0, 30),
        Position = UDim2.new(0, 10, 0, 7),
        BackgroundTransparency = 1,
        Parent = frame
    })

    local layout = Create("UIListLayout", {
        Padding = UDim.new(0, 10),
        FillDirection = Enum.FillDirection.Horizontal,
        SortOrder = Enum.SortOrder.LayoutOrder,
        VerticalAlignment = Enum.VerticalAlignment.Center,
        Parent = inner
    })

    local label = Create("TextLabel", {
        Text = name .. ":",
        Size = UDim2.new(0, 150, 1, 0),
        BackgroundTransparency = 1,
        TextColor3 = COLORS.TEXT,
        Font = Enum.Font.GothamBold,
        TextSize = 14,
        TextXAlignment = Enum.TextXAlignment.Left,
        LayoutOrder = 1,
        Parent = inner
    })

    local inputBox = Create("TextBox", {
        Text = tostring(init),
        Size = UDim2.new(0, 50, 1, 0),
        BackgroundColor3 = Color3.fromRGB(20, 20, 25),
        TextColor3 = COLORS.SELECTED,
        Font = Enum.Font.GothamBold,
        TextSize = 14,
        LayoutOrder = 2,
        Parent = inner
    })
    CreateCorner(4, inputBox)
    CreateStroke(1, Color3.fromRGB(60, 60, 65), inputBox)

    Bind(inputBox.FocusLost, function(enter)
        local num = tonumber(inputBox.Text)
        if num then
            num = math.clamp(num, min, max)
            inputBox.Text = tostring(num)
            callback(num)
        else
             local default = min
             inputBox.Text = tostring(default)
             callback(default)
        end
    end)
end

local OriginalFlingPos = nil
local OriginalCamSubject = nil

CreateCombatToggle("Fling", function(toggle)
    if toggle then
        FlingEnabled = not FlingEnabled
        local Hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid")
        local Root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        local Cam = workspace.CurrentCamera

        if FlingEnabled then
            if Root then OriginalFlingPos = Root.CFrame end
            if Cam then
                OriginalCamSubject = Cam.CameraSubject
                if FlingTarget and FlingTarget.Character and FlingTarget.Character:FindFirstChild("Humanoid") then
                    Cam.CameraSubject = FlingTarget.Character.Humanoid
                end
            end
            if LocalPlayer.Character then
                local hl = Instance.new("Highlight")
                hl.Name = "SelfFlingESP"
                hl.FillColor = Color3.fromRGB(0, 255, 100)
                hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                hl.FillTransparency = 0.5
                hl.Parent = LocalPlayer.Character
            end
        else
            if Hum then Hum.PlatformStand = false end
            if LocalPlayer.Character then
                for _, part in pairs(LocalPlayer.Character:GetChildren()) do
                    if part:IsA("BasePart") then part.CanCollide = true end
                end
            end
            if Root and OriginalFlingPos then
                Root.CFrame = OriginalFlingPos + Vector3.new(0, 2, 0)
                Root.AssemblyAngularVelocity = Vector3.zero
                Root.AssemblyLinearVelocity = Vector3.zero
                OriginalFlingPos = nil
                if Hum then Hum:ChangeState(Enum.HumanoidStateType.GettingUp) end
            end
            if Cam and OriginalCamSubject then
                Cam.CameraSubject = OriginalCamSubject
                OriginalCamSubject = nil
            elseif Cam and Hum then
                Cam.CameraSubject = Hum
            end
            if LocalPlayer.Character then
                local hl = LocalPlayer.Character:FindFirstChild("SelfFlingESP")
                if hl then hl:Destroy() end
            end
        end
    end
    return FlingEnabled
end, Color3.fromRGB(255, 50, 50))

local NoRagdollEnabled = false
local NoStunEnabled = false

RunService.Stepped:Connect(function()
    if not LocalPlayer.Character then return end
    local hum = LocalPlayer.Character:FindFirstChild("Humanoid")
    if not hum then return end

    if NoRagdollEnabled then
        hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
        hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
        if LocalPlayer.Character:GetAttribute("Ragdoll") then LocalPlayer.Character:SetAttribute("Ragdoll", nil) end
        if LocalPlayer.Character:GetAttribute("Ragdolled") then LocalPlayer.Character:SetAttribute("Ragdolled", nil) end
        if LocalPlayer.Character:GetAttribute("Knocked") then LocalPlayer.Character:SetAttribute("Knocked", nil) end
        for _, name in ipairs({"Ragdoll", "Ragdolled", "Knock", "Knocked", "RealRagdoll", "RagdollConstraints"}) do
             local obj = LocalPlayer.Character:FindFirstChild(name)
             if obj then obj:Destroy() end
        end
    end

    if NoStunEnabled then
        hum.PlatformStand = false
        hum.Sit = false
        hum:SetStateEnabled(Enum.HumanoidStateType.PlatformStanding, false)
        if LocalPlayer.Character:GetAttribute("Stunned") then LocalPlayer.Character:SetAttribute("Stunned", nil) end
        if LocalPlayer.Character:GetAttribute("Stun") then LocalPlayer.Character:SetAttribute("Stun", nil) end
        for _, name in ipairs({"Stun", "Stunned", "StunVar", "Action", "Freeze", "Slow"}) do
             local obj = LocalPlayer.Character:FindFirstChild(name)
             if obj then obj:Destroy() end
        end
    end
end)

-- [[ FLING SYSTEM ]] --
local FlingBAV = nil
local FlingBV = nil

Bind(RunService.Stepped, function()
    if FlingEnabled and FlingTarget and FlingTarget.Character and FlingTarget.Character:FindFirstChild("HumanoidRootPart") then
        local targetRoot = FlingTarget.Character.HumanoidRootPart
        local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        local myHum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid")

        if myRoot and myHum and myHum.Health > 0 then
            if not FlingBAV or FlingBAV.Parent ~= myRoot then
                 if FlingBAV then FlingBAV:Destroy() end
                 FlingBAV = Instance.new("BodyAngularVelocity")
                 FlingBAV.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
                 FlingBAV.Parent = myRoot
            end
            if not FlingBV or FlingBV.Parent ~= myRoot then
                 if FlingBV then FlingBV:Destroy() end
                 FlingBV = Instance.new("BodyVelocity")
                 FlingBV.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                 FlingBV.Parent = myRoot
            end
            myHum.PlatformStand = true
            local jitter = Vector3.new((math.random() - 0.5) * 4, -3.5 + (math.random() - 0.5) * 2, (math.random() - 0.5) * 4)
            local prediction = targetRoot.AssemblyLinearVelocity * 0.2
            myRoot.CFrame = CFrame.new(targetRoot.Position + prediction + jitter) * CFrame.Angles(math.random(-10, 10), math.random(-10, 10), math.random(-10, 10))
            myRoot.AssemblyLinearVelocity = targetRoot.AssemblyLinearVelocity
            FlingBV.Velocity = targetRoot.AssemblyLinearVelocity
            FlingBAV.AngularVelocity = Vector3.new(10000, 10000, 10000)
            local Cam = workspace.CurrentCamera
            if Cam and Cam.CameraSubject ~= targetRoot.Parent:FindFirstChild("Humanoid") then
                Cam.CameraSubject = targetRoot.Parent:FindFirstChild("Humanoid")
            end
        else
             if FlingBAV then FlingBAV:Destroy(); FlingBAV = nil end
             if FlingBV then FlingBV:Destroy(); FlingBV = nil end
        end
    else
        if FlingBAV then FlingBAV:Destroy(); FlingBAV = nil end
        if FlingBV then FlingBV:Destroy(); FlingBV = nil end
    end
end)

local LastAttack = 0
Bind(RunService.Heartbeat, function()
    if tick() - LastAttack < AttackDelay then return end
    LastAttack = tick()
    if (KillAura or KillAuraTP) and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        local myRoot = LocalPlayer.Character.HumanoidRootPart
        local myPos = myRoot.Position
        local Targets = {}
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
                local targetRoot = p.Character.HumanoidRootPart
                local targetDist = (targetRoot.Position - myPos).Magnitude
                local canAttack = false
                if KillAuraTP and p == SelectedPlayer then
                     myRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 0, 3) 
                     canAttack = true
                     local Cam = workspace.CurrentCamera
                     if Cam and Cam.CameraSubject ~= targetRoot.Parent:FindFirstChild("Humanoid") then Cam.CameraSubject = targetRoot.Parent:FindFirstChild("Humanoid") end
                elseif FlingEnabled and p == FlingTarget then
                     canAttack = true
                elseif KillAura and targetDist < KillRadius then
                     canAttack = true
                end
                if canAttack then table.insert(Targets, p.Character) end
            end
        end
        if #Targets > 0 then
            local eventsFolder = ReplicatedStorage:FindFirstChild("Events")
            if eventsFolder then
                local ce = eventsFolder:FindFirstChild("CombatEvent")
                local he = eventsFolder:FindFirstChild("HitEvent")
                if ce then pcall(function() ce:FireServer("Attack") end) end
                if he then pcall(function() he:FireServer(Targets) end) end
            end
            local Tool = LocalPlayer.Character:FindFirstChildOfClass("Tool")
            for _, targetChar in pairs(Targets) do
                if TargetRemoteName ~= "" and TargetRemoteName ~= "CombatEvent" and TargetRemoteName ~= "HitEvent" then
                    local customRemote = ReplicatedStorage:FindFirstChild(TargetRemoteName, true)
                    if customRemote then pcall(function() customRemote:FireServer(targetChar) end) end
                end
            end
            if Tool then Tool:Activate() end
        end
    end
end)

local minSpeed, maxSpeed = 16, 1000
local CurrentWalkSpeed, CurrentFlySpeed, WalkSpeedEnabled = 24, 24, false

local function CreateSlider(name, layoutOrder, initValue, callback)
    local section = Create("Frame", { Size = UDim2.new(1, 0, 0, 45), BackgroundTransparency = 1, LayoutOrder = layoutOrder, Parent = SpeedSection })
    CreateCorner(2, Create("Frame", { Size = UDim2.new(0, 4, 0, 4), Position = UDim2.new(0, 0, 0, 10), BackgroundColor3 = COLORS.SELECTED, Parent = section }))
    local inner = Create("Frame", { Size = UDim2.new(1, -15, 0, 20), Position = UDim2.new(0, 10, 0, 2), BackgroundTransparency = 1, Parent = section })
    Create("UIListLayout", { Padding = UDim.new(0, 10), FillDirection = Enum.FillDirection.Horizontal, SortOrder = Enum.SortOrder.LayoutOrder, Parent = inner })
    Create("TextLabel", { Text = name, Size = UDim2.new(0, 90, 1, 0), BackgroundTransparency = 1, TextColor3 = COLORS.TEXT, Font = Enum.Font.GothamBold, TextSize = 13, Parent = inner })
    local InputBox = Create("TextBox", { Text = tostring(initValue), Size = UDim2.new(0, 40, 1, 0), BackgroundColor3 = COLORS.BG, TextColor3 = COLORS.SELECTED, Font = Enum.Font.GothamBold, TextSize = 13, Parent = inner })
    CreateCorner(4, InputBox)
    CreateStroke(1, Color3.fromRGB(60, 60, 65), InputBox)
    Bind(InputBox.FocusLost, function()
        local num = tonumber(InputBox.Text)
        if num then num = math.clamp(num, minSpeed, maxSpeed) InputBox.Text = tostring(num) callback(num) else InputBox.Text = tostring(minSpeed) callback(minSpeed) end
    end)
    return inner, InputBox
end

local function UpdateToggle(btn, knob, stroke, state)
    local targetPos = state and UDim2.new(1, -18, 0.5, -8) or UDim2.new(0, 2, 0.5, -8)
    local targetColor = state and COLORS.SELECTED or COLORS.BUTTON
    local strokeColor = state and Color3.new(1, 1, 1) or Color3.fromRGB(80, 80, 80)
    TweenService:Create(knob, TweenInfo.new(0.2), {Position = targetPos}):Play()
    TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = targetColor}):Play()
    TweenService:Create(stroke, TweenInfo.new(0.2), {Color = strokeColor}):Play()
end

-- [[ LOOP: FLY & SPEED ]] --
RunService.RenderStepped:Connect(function()
    local Char = LocalPlayer.Character
    local Hum = Char and Char:FindFirstChild("Humanoid")
    local Root = Char and Char:FindFirstChild("HumanoidRootPart")
    if Char and Hum and Root and Hum.Health > 0 then
        if WalkSpeedEnabled then Hum.WalkSpeed = CurrentWalkSpeed end
        if IsFlying then
            Hum.PlatformStand = true
            local Cam = workspace.CurrentCamera
            if Cam then
                local moveDir = Vector3.zero
                local forward, side = 0, 0
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then forward = 1 end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then forward = -1 end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then side = -1 end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then side = 1 end
                if forward ~= 0 or side ~= 0 then
                    moveDir = (Cam.CFrame.LookVector * forward) + (Cam.CFrame.RightVector * side)
                    Root.AssemblyLinearVelocity = moveDir.Unit * CurrentFlySpeed
                else
                    Root.AssemblyLinearVelocity = Vector3.zero
                end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then Root.AssemblyLinearVelocity = Root.AssemblyLinearVelocity + Vector3.new(0, CurrentFlySpeed, 0) end
                Root.AssemblyAngularVelocity = Vector3.zero
            end
        else
            if Hum.PlatformStand and not NoStunEnabled then Hum.PlatformStand = false Root.AssemblyAngularVelocity = Vector3.zero end
        end
    end
end)

local function UpdateESP()
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local hl = player.Character:FindFirstChild("ViltrumitesESP")
            if player == SelectedPlayer then
                if not hl then hl = Create("Highlight", { Name = "ViltrumitesESP", FillColor = Color3.fromRGB(255, 40, 40), Parent = player.Character }) end
            else if hl then hl:Destroy() end end
        end
    end
end
Bind(RunService.Heartbeat, UpdateESP)

local function RefreshPlayers()
    for _, child in pairs(PlayerContainer:GetChildren()) do if child:IsA("TextButton") then child:Destroy() end end
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local btn = Create("TextButton", { 
                Size = UDim2.new(1, 0, 0, 45), 
                BackgroundColor3 = (SelectedPlayer == player) and COLORS.SELECTED or COLORS.BUTTON, 
                Parent = PlayerContainer, 
                Text = "" 
            })
            CreateCorner(6, btn)
            local avatar = Create("ImageLabel", { 
                Size = UDim2.new(0, 35, 0, 35), 
                Position = UDim2.new(0, 5, 0.5, -17), 
                BackgroundTransparency = 0.5, 
                Parent = btn 
            })
            CreateCorner(18, avatar)
            Create("TextLabel", { 
                Text = player.DisplayName, 
                Size = UDim2.new(1, -55, 1, 0), 
                Position = UDim2.new(0, 50, 0, 0), 
                BackgroundTransparency = 1, 
                TextColor3 = COLORS.TEXT, 
                Font = Enum.Font.GothamBold, 
                TextSize = 15, 
                TextXAlignment = Enum.TextXAlignment.Left,
                Parent = btn 
            })
            task.spawn(function() pcall(function() avatar.Image = Players:GetUserThumbnailAsync(player.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size420x420) end) end)
            Bind(btn.MouseButton1Click, function()
                if SelectedPlayer == player then
                    SelectedPlayer = nil
                    FlingTarget = nil
                else
                    SelectedPlayer = player
                    FlingTarget = player
                end
                RefreshPlayers()
            end)
        end
    end
end
Bind(Players.PlayerAdded, RefreshPlayers)
Bind(Players.PlayerRemoving, RefreshPlayers)
RefreshPlayers()

-- [[ CHARACTER BYPASS ]] --
local function ChangeCharacter(charName)
    if charName ~= "BigBrain" and charName ~= "AtomMistress" then
        local Event = ReplicatedStorage:FindFirstChild("Events") and ReplicatedStorage.Events:FindFirstChild("ServerEvent")
        if Event then Event:FireServer("ChangeCharacter", charName) end
    end
    pcall(function() LocalPlayer:SetAttribute("Character", charName) end)
    task.delay(0.5, function() if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then LocalPlayer.Character.Humanoid.Health = 0 end end)
end

local function FullCleanup()
    for _, con in pairs(ConnectionTable) do con:Disconnect() end
    -- Cleanup Global ESPs
    for _, player in pairs(Players:GetPlayers()) do
        if player.Character then
            local hl = player.Character:FindFirstChild("ViltrumitesESP") or player.Character:FindFirstChild("SelfFlingESP")
            if hl then hl:Destroy() end
        end
    end
    -- Cleanup Workspace (NPC ESP if any left)
    for _, obj in pairs(workspace:GetChildren()) do
        local hl = obj:FindFirstChild("ViltrumitesNPCESP")
        if hl then hl:Destroy() end
    end
    if ScreenGui then ScreenGui:Destroy() end
    if OpenBtn then OpenBtn:Destroy() end
    _G[SCRIPT_ID] = nil
end
_G[SCRIPT_ID] = { Cleanup = FullCleanup }
Bind(CloseBtn.MouseButton1Click, FullCleanup)
