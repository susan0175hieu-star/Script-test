-- [[ UI LIBRARY HELPER ]] --
local function Create(className, props)
    local inst = Instance.new(className)
    for k, v in pairs(props) do
        inst[k] = v
    end
    return inst
end

local function CreateCorner(radius, parent)
    return Create("UICorner", { CornerRadius = UDim.new(0, radius), Parent = parent })
end

local function CreateStroke(thickness, color, parent)
    return Create("UIStroke", { Thickness = thickness, Color = color, ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Parent = parent })
end

local function AddGradient(parent, color1, color2)
    return Create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, color1),
            ColorSequenceKeypoint.new(1, color2)
        }),
        Rotation = 45,
        Parent = parent
    })
end

local function Bind(signal, callback)
    local con = signal:Connect(callback)
    table.insert(ConnectionTable, con)
    return con
end

local function CreateCombatToggle(text, callback, activeColor)
    activeColor = activeColor or COLORS.SELECTED
    local frame = Create("Frame", {
        Size = UDim2.new(1, 0, 0, 45),
        BackgroundColor3 = COLORS.BUTTON,
        Parent = CombatContainer
    })
    CreateCorner(6, frame)
    local stroke = CreateStroke(1, Color3.fromRGB(70, 70, 75), frame)
    
    local label = Create("TextLabel", {
        Text = text,
        Size = UDim2.new(1, -70, 1, 0),
        Position = UDim2.new(0, 15, 0, 0),
        BackgroundTransparency = 1,
        TextColor3 = COLORS.TEXT,
        Font = Enum.Font.GothamBold,
        TextSize = 14,
        TextXAlignment = Enum.TextXAlignment.Left,
        Parent = frame
    })
    
    -- Switch Background
    local switchBG = Create("Frame", {
        Size = UDim2.new(0, 44, 0, 22),
        Position = UDim2.new(1, -55, 0.5, -11),
        BackgroundColor3 = Color3.fromRGB(30, 30, 35),
        Parent = frame
    })
    CreateCorner(11, switchBG)
    local switchStroke = CreateStroke(2, Color3.fromRGB(50, 50, 55), switchBG)
    
    -- Knob
    local knob = Create("Frame", {
        Size = UDim2.new(0, 18, 0, 18),
        Position = UDim2.new(0, 2, 0.5, -9),
        BackgroundColor3 = Color3.fromRGB(180, 180, 180),
        Parent = switchBG
    })
    CreateCorner(10, knob)
    
    local btn = Create("TextButton", {
        Text = "",
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        Parent = frame
    })
    
    local function Update(animate)
        local state = (typeof(callback(nil)) == "boolean") and callback(nil) or false
        local targetPos = state and UDim2.new(1, -20, 0.5, -9) or UDim2.new(0, 2, 0.5, -9)
        local targetColor = state and COLORS.SELECTED or Color3.fromRGB(30, 30, 35)
        local knobColor = state and Color3.new(1, 1, 1) or Color3.fromRGB(180, 180, 180)
        
        if animate then
            TweenService:Create(knob, TweenInfo.new(0.2, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Position = targetPos, BackgroundColor3 = knobColor}):Play()
            TweenService:Create(switchBG, TweenInfo.new(0.2, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {BackgroundColor3 = targetColor}):Play()
            TweenService:Create(switchStroke, TweenInfo.new(0.2, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Color = state and (activeColor == COLORS.SELECTED and Color3.new(1, 1, 1) or activeColor) or Color3.fromRGB(50, 50, 55), Transparency = state and 0.5 or 0}):Play()
            TweenService:Create(stroke, TweenInfo.new(0.2), {Color = state and activeColor or Color3.fromRGB(70, 70, 75)}):Play()
        else
            knob.Position = targetPos
            knob.BackgroundColor3 = knobColor
            switchBG.BackgroundColor3 = targetColor
            switchStroke.Color = state and (activeColor == COLORS.SELECTED and Color3.new(1, 1, 1) or activeColor) or Color3.fromRGB(50, 50, 55)
            switchStroke.Transparency = state and 0.5 or 0
            stroke.Color = state and activeColor or Color3.fromRGB(70, 70, 75)
        end
    end
    
    Bind(btn.MouseButton1Click, function()
        callback(true)
        Update(true)
    end)
    Update(false)
end

local function CreateCombatSlider(name, min, max, init, callback)
    local frame = Create("Frame", {
        Size = UDim2.new(1, 0, 0, 45),
        BackgroundTransparency = 1,
        Parent = CombatContainer
    })
    
    local dot = Create("Frame", {
        Size = UDim2.new(0, 4, 0, 4),
        Position = UDim2.new(0, 0, 0, 8),
        BackgroundColor3 = COLORS.SELECTED,
        Parent = frame
    })
    CreateCorner(2, dot)
    
    -- Inner Content Frame (for Auto Layout)
    local inner = Create("Frame", {
        Size = UDim2.new(1, -15, 0, 20), -- Height matches input/label
        Position = UDim2.new(0, 10, 0, 0),
        BackgroundTransparency = 1,
        Parent = frame
    })
    
    local layout = Create("UIListLayout", {
        Padding = UDim.new(0, 10),
        FillDirection = Enum.FillDirection.Horizontal,
        SortOrder = Enum.SortOrder.LayoutOrder,
        VerticalAlignment = Enum.VerticalAlignment.Center,
        Parent = inner
    })
    
    -- Label (Auto Sized)
    local label = Create("TextLabel", {
        Text = name .. ":",
        Size = UDim2.new(0, 120, 1, 0), -- Fixed Width for alignment
        -- AutomaticSize = Enum.AutomaticSize.X,
        BackgroundTransparency = 1,
        TextColor3 = COLORS.TEXT,
        Font = Enum.Font.GothamBold,
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Left,
        LayoutOrder = 1,
        Parent = inner
    })
    
    -- Input Box (Flows after label)
    local inputBox = Create("TextBox", {
        Text = tostring(init),
        Size = UDim2.new(0, 50, 1, 0),
        BackgroundColor3 = Color3.fromRGB(20, 20, 25),
        TextColor3 = COLORS.SELECTED,
        Font = Enum.Font.GothamBold,
        TextSize = 13,
        LayoutOrder = 2,
        Parent = inner
    })
    CreateCorner(4, inputBox)
    local inputStroke = CreateStroke(1, Color3.fromRGB(60, 60, 65), inputBox)
    
    -- Sync Input Box
    Bind(inputBox.FocusLost, function(enter)
        local num = tonumber(inputBox.Text)
        if num then
            num = math.clamp(num, min, max)
            inputBox.Text = tostring(num) -- VISUAL UPDATE
            callback(num)
        else
             local default = min
             inputBox.Text = tostring(default)
             callback(default)
        end
    end)
end

local NoRagdollEnabled = false
-- Toggle moved to Speed Section

local NoStunEnabled = false
-- Toggle moved to Speed Section

-- Loop for No Ragdoll and No Stun
RunService.Stepped:Connect(function()
    if not LocalPlayer.Character then return end
    local hum = LocalPlayer.Character:FindFirstChild("Humanoid")
    if not hum then return end

    if NoRagdollEnabled then
        hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
        hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
        
        -- Remove Ragdoll Attributes
        if LocalPlayer.Character:GetAttribute("Ragdoll") then LocalPlayer.Character:SetAttribute("Ragdoll", nil) end
        if LocalPlayer.Character:GetAttribute("Ragdolled") then LocalPlayer.Character:SetAttribute("Ragdolled", nil) end
        if LocalPlayer.Character:GetAttribute("Knocked") then LocalPlayer.Character:SetAttribute("Knocked", nil) end
        
        -- Remove Ragdoll Objects
        for _, name in ipairs({"Ragdoll", "Ragdolled", "Knock", "Knocked", "RealRagdoll", "RagdollConstraints"}) do
             local obj = LocalPlayer.Character:FindFirstChild(name)
             if obj then obj:Destroy() end
        end
    end
    
    if NoStunEnabled then
        hum.PlatformStand = false
        hum.Sit = false
        hum:SetStateEnabled(Enum.HumanoidStateType.PlatformStanding, false)
        
        -- Remove Stun Attributes
        if LocalPlayer.Character:GetAttribute("Stunned") then LocalPlayer.Character:SetAttribute("Stunned", nil) end
        if LocalPlayer.Character:GetAttribute("Stun") then LocalPlayer.Character:SetAttribute("Stun", nil) end
        
        -- Remove Stun Objects
        for _, name in ipairs({"Stun", "Stunned", "StunVar", "Action", "Freeze", "Slow"}) do
             local obj = LocalPlayer.Character:FindFirstChild(name)
             if obj then obj:Destroy() end
        end
    end
end)

-- Instant Stun/Ragdoll Removal (ChildAdded)
local function SetupChildAdded(char)
    if not char then return end
    char.ChildAdded:Connect(function(child)
        if NoStunEnabled then
            if table.find({"Stun", "Stunned", "StunVar", "Action", "Freeze", "Slow"}, child.Name) then
                task.wait() 
                child:Destroy()
            end
        end
        
        if NoRagdollEnabled then
            if table.find({"Ragdoll", "Ragdolled", "Knock", "Knocked", "RealRagdoll", "RagdollConstraints"}, child.Name) then
                task.wait() 
                child:Destroy()
            end
        end
    end)
end

-- [[ FLING LOGIC ]] --
local BodyVelocity = Instance.new("BodyVelocity")
BodyVelocity.Velocity = Vector3.new(0, 0, 0)
BodyVelocity.MaxForce = Vector3.new(math.huge, math.huge, math.huge)

local BodyAngularVelocity = Instance.new("BodyAngularVelocity")
BodyAngularVelocity.AngularVelocity = Vector3.new(0, 99999, 0)
BodyAngularVelocity.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)

Bind(RunService.Stepped, function()
    local hasTarget = FlingTarget and FlingTarget.Character and FlingTarget.Character:FindFirstChild("HumanoidRootPart")
    
    -- SAFETY CHECK: If we fall into the void (Countered/BgBug), disable Fling immediately
    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        if LocalPlayer.Character.HumanoidRootPart.Position.Y < -500 then
            FlingEnabled = false
            
            -- Reset Physics
            LocalPlayer.Character.HumanoidRootPart.AssemblyLinearVelocity = Vector3.zero
            LocalPlayer.Character.HumanoidRootPart.AssemblyAngularVelocity = Vector3.zero
            
            if LocalPlayer.Character:FindFirstChild("Humanoid") then
                LocalPlayer.Character.Humanoid.PlatformStand = false
                LocalPlayer.Character.Humanoid:ChangeState(Enum.HumanoidStateType.Dead) -- Force respawn
            end

            -- Auto Re-enable after 2 seconds (Respawn time)
            task.delay(2, function()
                FlingEnabled = true
            end)
            return
        end
    end
     
    -- NORMAL CHECKS
    if not FlingEnabled or not hasTarget or not LocalPlayer.Character or not LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then 
        BodyVelocity.Parent = nil
        BodyAngularVelocity.Parent = nil
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
            LocalPlayer.Character.Humanoid.PlatformStand = false
        end
        return 
    end
    
    local Root = LocalPlayer.Character.HumanoidRootPart
    local Hum = LocalPlayer.Character:FindFirstChild("Humanoid")
    
    -- Check if we died
    if Hum and Hum.Health <= 0 then
        FlingEnabled = false
        return
    end
    
    if Hum then Hum.PlatformStand = true end
    
    pcall(function()
        BodyVelocity.Parent = Root
        BodyAngularVelocity.Parent = Root
    end)
    
    local targetRoot = FlingTarget.Character.HumanoidRootPart
    
    -- Chaotic Fling: Lock position NEAR target (jitter) + Random Rotation
    -- Accuracy Tuned: Reduced multiplier to 4 for better hit consistency
    local jitter = Vector3.new(math.random() - 0.5, math.random() - 0.5, math.random() - 0.5) * 4
    Root.CFrame = CFrame.new(targetRoot.Position + jitter) * CFrame.Angles(math.random(-10, 10), math.random(-10, 10), math.random(-10, 10))
    Root.AssemblyLinearVelocity = Vector3.zero 
    BodyVelocity.Velocity = Vector3.zero -- Ensure no drift
    BodyAngularVelocity.AngularVelocity = Vector3.new(10000, 10000, 10000) -- Clean standard torque
end)

-- CAMERA STABILIZER (RenderStepped - High Priority)
Bind(RunService.RenderStepped, function()
    local Cam = workspace.CurrentCamera
    
    if FlingEnabled and FlingTarget and FlingTarget.Character and FlingTarget.Character:FindFirstChild("Humanoid") then
        -- SPECTATE MODE: Set CameraSubject to the Target's Humanoid
        -- This allows standard Roblox camera control AROUND the enemy
        -- The enemy is not spinning, so the camera is stable.
        if Cam.CameraSubject ~= FlingTarget.Character.Humanoid then
            Cam.CameraType = Enum.CameraType.Custom
            Cam.CameraSubject = FlingTarget.Character.Humanoid
        end
    elseif not FlingEnabled and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
         -- RESTORE: Set CameraSubject back to Self
         if Cam.CameraSubject ~= LocalPlayer.Character.Humanoid then
             Cam.CameraType = Enum.CameraType.Custom
             Cam.CameraSubject = LocalPlayer.Character.Humanoid
         end
    end
end)

-- KILL AOE Loop
local LastAttack = 0
Bind(RunService.Heartbeat, function()
    if tick() - LastAttack < AttackDelay then return end
    LastAttack = tick()
    
    if (KillAura or KillAuraTP) and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        local myRoot = LocalPlayer.Character.HumanoidRootPart
        local myPos = myRoot.Position
        
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
                local targetRoot = p.Character.HumanoidRootPart
                local targetDist = (targetRoot.Position - myPos).Magnitude
                
                -- Standard Aura (Distance Check) OR TP Aura (Always hits if within slightly larger range or purely selected)
                -- For safety, we keep TP range somewhat limited or logic based
                
                local canAttack = false
                
                -- AUTO TP: Only target the SELECTED player
                if KillAuraTP and p == SelectedPlayer then
                     -- TELEPORT BEHIND TARGET
                     myRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 0, 3) 
                     canAttack = true
                
                -- AUTO FARM/AURA: Distance check for everyone else
                elseif KillAura and targetDist < KillRadius then
                     canAttack = true
                end
                
                if canAttack then
                    -- Attack target p
                    local targetChar = p.Character
                    
                    -- 1. Try Remote Event (Based on Sniffer Data)
                    local eventsFolder = ReplicatedStorage:FindFirstChild("Events")
                    if eventsFolder then
                        local ce = eventsFolder:FindFirstChild("CombatEvent")
                        local he = eventsFolder:FindFirstChild("HitEvent")
                        
                        -- Pattern found in Sniffer:
                        -- CombatEvent fires with "Attack"
                        -- HitEvent fires with a Table containing the target
                        if ce then pcall(function() ce:FireServer("Attack") end) end
                        if he then pcall(function() he:FireServer({ targetChar }) end) end
                    end

                    -- 2. Fallback User-defined Remote
                    if TargetRemoteName ~= "" and TargetRemoteName ~= "CombatEvent" and TargetRemoteName ~= "HitEvent" then
                        local customRemote = ReplicatedStorage:FindFirstChild(TargetRemoteName, true)
                        if customRemote then
                            pcall(function() customRemote:FireServer(targetChar) end)
                        end
                    end
                    
                    -- 2. Tool Activate / Click (Melee Range Only)
                    if targetDist < 15 or KillAuraTP then 
                        local Tool = LocalPlayer.Character:FindFirstChildOfClass("Tool")
                        if Tool then Tool:Activate() end
                    end
                end
            end
        end
    end
end)

-- OPTIMIZED SPEED & FLY LOOP (Heartbeat)
Bind(RunService.RenderStepped, function()
    local Char = LocalPlayer.Character
    local Hum = Char and Char:FindFirstChild("Humanoid")
    local Root = Char and Char:FindFirstChild("HumanoidRootPart")
    
    if Char and Hum and Root and Hum.Health > 0 then
        -- 1. Apply WalkSpeed
        if WalkSpeedEnabled then
            Hum.WalkSpeed = CurrentWalkSpeed
        end
        
        -- 2. Apply Fly
        if IsFlying then
            Hum.PlatformStand = true -- Disable physics/animations
            local Cam = workspace.CurrentCamera
            if Cam then
                local moveDir = Vector3.zero
                local look = Cam.CFrame.LookVector
                local right = Cam.CFrame.RightVector
                
                -- Manual Movement Keys (Works in PlatformStand)
                local forward = 0
                local side = 0
                
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then forward = 1 end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then forward = -1 end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then side = -1 end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then side = 1 end
                
                if forward ~= 0 or side ~= 0 then
                    moveDir = (look * forward) + (right * side)
                    if moveDir.Magnitude > 0 then
                        Root.AssemblyLinearVelocity = moveDir.Unit * CurrentFlySpeed
                    end
                else
                    -- Stop if no keys pressed
                    Root.AssemblyLinearVelocity = Vector3.zero 
                end
                
                -- Optional Vertical Fly
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then 
                    Root.AssemblyLinearVelocity = Root.AssemblyLinearVelocity + Vector3.new(0, CurrentFlySpeed, 0)
                end

                
                -- STABILIZE ROTATION (Prevent Spinning)
                Root.AssemblyAngularVelocity = Vector3.zero
            end
        else
             -- Ensure PlatformStand is disabled if not flying (Safety Check)
             if Hum.PlatformStand and not NoStunEnabled then
                 Hum.PlatformStand = false
                 Root.AssemblyAngularVelocity = Vector3.zero -- Stop spinning on land
             end
        end


    end
end)

-- [[ LOGIC: CHARACTER BYPASS ]] --
local function ChangeCharacter(charName)
    CurrentFakeChar = charName
    
    -- 1. Server Event (RESTORED DEFAULT MECHANISM)
    -- This allows free characters (Mark, PowerPlex, TechSuit) to switch instantly/normally
    -- OPTIMIZATION: Skip FireServer for BigBrain/Paid chars to avoid server-side checks/purchase prompts
    if charName ~= "BigBrain" and charName ~= "AtomMistress" then
        local args = {
            [1] = "ChangeCharacter",
            [2] = charName
        }
        local Event = ReplicatedStorage:FindFirstChild("Events") and ReplicatedStorage.Events:FindFirstChild("ServerEvent")
        if Event then Event:FireServer(unpack(args)) end
    end

    -- 2. Client Bypass (Attribute) - Keeps working as backup/for paid chars
    pcall(function() LocalPlayer:SetAttribute("Character", charName) end)
    
    -- 3. Force Respawn with Delay (Like Big Brain/Atom Mistress)
    -- This allows the server to process the attribute change before we die
    task.delay(0.5, function()
         local Char = LocalPlayer.Character
         if Char and Char:FindFirstChild("Humanoid") then 
             Char.Humanoid.Health = 0 
         end
    end)
end
